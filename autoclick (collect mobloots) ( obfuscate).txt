-- Detecting Position
local currentDetectingSlot = nil

UIS.InputBegan:Connect(function(input, gameProcessed)
    if currentDetectingSlot and (input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1) then
        local pos = Vector2.new(input.Position.X, input.Position.Y)
        currentDetectingSlot.position = pos
        currentDetectingSlot.xLabel.Text = "X: " .. math.floor(pos.X)
        currentDetectingSlot.yLabel.Text = "Y: " .. math.floor(pos.Y)
        currentDetectingSlot.detectButton.Text = "Detect " .. currentDetectingSlot.slotLabel.Text:sub(-1)
        currentDetectingSlot = nil
    end
end)

for i, slot in ipairs(slots) do
    slot.detectButton.MouseButton1Click:Connect(function()
        currentDetectingSlot = slot
        slot.detectButton.Text = "Listening..."
    end)
end

-- Auto Click System

local function rapidClick(pos, duration)
    if pos ~= Vector2.new(0, 0) then
        local clicks = 5  -- Number of rapid clicks to perform
        for _ = 1, clicks do
            VirtualInputManager:SendMouseButtonEvent(pos.X + 30, pos.Y + 50, 0, true, game, 1)
            VirtualInputManager:SendMouseButtonEvent(pos.X + 30, pos.Y + 50, 0, false, game, 1)
            task.wait(duration)
        end
    end
end

autoClickButton.MouseButton1Click:Connect(function()
    isRunning = not isRunning
    autoClickButton.Text = isRunning and "STOP AUTO CLICK" or "START AUTO CLICK"
    
    if isRunning then
        task.spawn(function()
            while isRunning do
                for i = 1, 3 do
                    if slots[i].position ~= Vector2.new(0, 0) then
                        rapidClick(slots[i].position, tonumber(slots[i].delayBox.Text) or 0.1)
                    end
                end
                
                -- Universal delay
                task.wait(tonumber(universalDelayBox.Text) or 0.1)
            end
        end)
    end
end)
