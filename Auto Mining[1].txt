local function teleportAndHarvest(location)
    -- Teleport to the location
    game.Players.LocalPlayer.Character.HumanoidRootPart.CFrame = CFrame.new(location.x, location.y, location.z)
    
    -- Wait 0.5 seconds after teleportation
    wait(0.5)
    
    -- Harvest
    local args = {
        [1] = "Harvest",
        [2] = location.id
    }
    HarvestEvent:FireServer(unpack(args))
    
    -- Wait 8 seconds before moving to the next location
    wait(8)
end

while true do
    for _, location in ipairs(locations) do
        teleportAndHarvest(location)
    end
end
